23def read_snapshot(path): 

24 """Read one live JSON snapshot and tolerate files being replaced concurrently.""" 

25 try: 

26 return json.loads(path.read_text(encoding='utf-8')) 

27 except (OSError, json.JSONDecodeError): 

28 return None

31def food_eaten(snapshot, side): 

32 """Use recorded food counts and avoid inferring full growth from partial cube views.""" 

33 stored = snapshot.get('food_eaten') or {} 

34 if side in stored: 

35 return stored[side] 

36 if int((snapshot.get('turn_data') or {}).get('levels', 1)) > 1: 

37 return None 

38 return flat_food_estimate(snapshot, side)

41def flat_food_estimate(snapshot, side): 

42 """Estimate historical growth only when a complete flat board is available.""" 

43 board = str((snapshot.get('turn_data') or {}).get('board', '')) 

44 return max(0, sum(char in (side, side.lower()) for char in board) - INITIAL_SNAKE_LENGTH)

47def game_summary(snapshot): 

48 """Build the compact match-list schema from a public live snapshot.""" 

49 data = snapshot.get('turn_data') or {} 

50 return { 

51 'game_id': snapshot.get('game_id', ''), 

52 'status': snapshot.get('status', 'playing'), 

53 'started_at': snapshot.get('started_at', snapshot.get('updated_at', 0)), 

54 'updated_at': snapshot.get('updated_at', 0), 

55 'turn_number': snapshot.get('turn_number', 0), 

56 'bot_version': snapshot.get('bot_version', ''), 

57 'player_1': data.get('player_1', 'Jugador A'), 

58 'player_2': data.get('player_2', 'Jugador B'), 

59 'score_1': data.get('score_1', 0), 

60 'score_2': data.get('score_2', 0), 

61 'remaining_moves': data.get('remaining_moves', 0), 

62 'side': data.get('side', ''), 

63 'food_eaten_1': food_eaten(snapshot, 'A'), 

64 'food_eaten_2': food_eaten(snapshot, 'B'),

68class ViewerHandler(SimpleHTTPRequestHandler): 

69 """Serve local viewer assets and read-only match APIs without exposing bot credentials.""" 

70 

71 def __init__(self, *args, **kwargs): 

72 super().__init__(*args, directory=str(VIEWER_DIR), **kwargs)

74 def end_headers(self): 

75 """Disable caching and allow only the configured browser origins.""" 

76 self.send_header('Cache-Control', 'no-store') 

77 origin = self.headers.get('Origin', '') 

78 if origin in ALLOWED_BROWSER_ORIGINS: 

79 self.send_header('Access-Control-Allow-Origin', origin) 

80 self.send_header('Access-Control-Allow-Methods', 'GET, OPTIONS') 

81 self.send_header('Access-Control-Allow-Headers', 'Content-Type') 

82 self.send_header('Access-Control-Allow-Private-Network', 'true') 

83 self.send_header('Vary', 'Origin') 

84 super().end_headers() 

85 

86 def do_OPTIONS(self): 

87 """Answer read-only browser preflight requests.""" 

88 self.send_response(204) 

89 self.end_headers() 

90 

91 def send_json(self, payload, status=200): 

92 """Send a UTF-8 JSON response with explicit status and content length.""" 

93 body = json.dumps(payload).encode('utf-8') 

94 self.send_response(status) 

95 self.send_header('Content-Type', 'application/json; charset=utf-8') 

96 self.send_header('Content-Length', str(len(body))) 

97 self.end_headers() 

98 self.wfile.write(body)

100 def do_GET(self): 

101 """Route match APIs and delegate static assets to the standard file handler.""" 

102 path = urlparse(self.path).path 

103 if path == '/api/games': 

104 return self.send_json({'games': recent_games()}) 

105 if path.startswith('/api/game/'): 

106 return self.send_game(unquote(path.removeprefix('/api/game/'))) 

107 if path == '/api/health': 

108 return self.send_json({'ok': True}) 

109 return super().do_GET() 

110 

111 def send_game(self, game_id): 

112 """Validate a match identifier and return its public snapshot or a JSON error.""" 

113 if not valid_game_id(game_id): 

114 return self.send_json({'error': 'invalid game id'}, 400) 

115 snapshot = read_snapshot(LIVE_DIR / f'game_{game_id}.json') 

116 if snapshot is None: 

117 return self.send_json({'error': 'game not found'}, 404) 

118 return self.send_json(snapshot)

120 def log_message(self, fmt, *args): 

121 """Suppress routine successful requests while retaining other server diagnostics.""" 

122 if args and str(args[1]) == '200': 

123 return 

124 super().log_message(fmt, *args)

127def valid_game_id(game_id): 

128 """Restrict snapshot identifiers to safe filename characters.""" 

129 return bool(game_id) and all(char.isalnum() or char in '-_' for char in game_id)

132def recent_games(): 

133 """Read valid snapshots and order the bounded match list by activity and status.""" 

134 if not LIVE_DIR.exists(): 

135 return [] 

136 summaries = [] 

137 for path in LIVE_DIR.glob('game_*.json'): 

138 snapshot = read_snapshot(path) 

139 if snapshot: 

140 summaries.append(game_summary(snapshot)) 

141 return sorted(summaries, key=game_sort_key)[:100]

144def game_sort_key(game): 

145 """Keep live games first and use the appropriate start or finish timestamp.""" 

146 playing = game['status'] == 'playing' 

147 timestamp = game['started_at'] if playing else game['updated_at'] 

148 return game['status'] == 'finished', timestamp

151def main(): 

152 """Start the loopback viewer server and release its socket when interrupted.""" 

153 parser = argparse.ArgumentParser(description='Live viewer for CodeChallenge Snake matches') 

154 parser.add_argument('--host', default='127.0.0.1') 

155 parser.add_argument('--port', type=int, default=8765) 

156 args = parser.parse_args() 

157 server = ThreadingHTTPServer((args.host, args.port), ViewerHandler) 

158 print(f'Snake live viewer: http://{args.host}:{args.port}') 

159 try: 

160 server.serve_forever() 

161 except KeyboardInterrupt: 

162 pass 

163 finally: 

164 server.server_close()

167if __name__ == '__main__': 

168 main()
