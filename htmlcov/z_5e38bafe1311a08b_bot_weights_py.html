178def load_weights(path): 

179 """Read a JSON configuration and merge recognized coefficients with defaults.""" 

180 raw = json.loads(Path(path).read_text()) 

181 weights = dict(DEFAULT_WEIGHTS) 

182 weights.update({key: int(value) for key, value in raw.items() if key in DEFAULT_WEIGHTS}) 

183 return weights

187 """Load the optional local configuration and report whether a file was available.""" 

188 weight_path = Path(path)
